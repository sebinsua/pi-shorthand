// Calls first (the generic level last), then imports, then remove the deprecated module.
sg.rewrite('log("info", $M)', "logger.info($M)", "src/features");
sg.rewrite('log("warn", $M)', "logger.warn($M)", "src/features");
sg.rewrite('log("error", $M, $E)', "logger.error($M, { error: $E })", "src/features");
sg.rewrite("log($LEVEL, $M)", "logger.log($LEVEL, $M)", "src/features");
for (const file of glob("src/features/**/*.ts")) {
	const text = await Bun.file(file).text();
	const next = text.replace(/import \{ log(, type LogLevel)? \} from "((?:\.\.?\/)+lib)\/log";/, 'import { logger$1 } from "$2/logger";');
	if (next !== text) await Bun.write(file, next);
}
await $`rm src/lib/log.ts`;
